💻 代碼示例

以下是如何在您的網站中調用此API的示例代碼：

async function calculateAstroChart(birthData) { const response = await fetch('http://localhost:5000/api/calculate_chart', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(birthData) }); const data = await response.json(); if (data.success) { console.log('角色:', data.character); console.log('占星數據:', data.astro_data); return data; } else { console.error('錯誤:', data.error); return null; } }
import requests def calculate_astro_chart(birth_data): url = 'http://localhost:5000/api/calculate_chart' response = requests.post(url, json=birth_data) if response.status_code == 200: data = response.json() if data.get('success'): print('角色:', data['character']) return data return None # 使用示例 result = calculate_astro_chart({ "name": "測試用戶", "year": 1990, "month": 6, "day": 15, "hour": 14, "minute": 30, "city": "台北", "longitude": 121.55, "latitude": 25.017 })
<?php $birthData = array( "name" => "測試用戶", "year" => 1990, "month" => 6, "day" => 15, "hour" => 14, "minute" => 30, "city" => "台北", "longitude" => 121.55, "latitude" => 25.017 ); $ch = curl_init('http://localhost:5000/api/calculate_chart'); curl_setopt($ch, CURLOPT_RETURNTRANSFER, true); curl_setopt($ch, CURLOPT_POST, true); curl_setopt($ch, CURLOPT_HTTPHEADER, array('Content-Type: application/json')); curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($birthData)); $response = curl_exec($ch); curl_close($ch); $data = json_decode($response, true); if ($data['success']) { echo "角色: " . $data['character']['name']; } ?>
